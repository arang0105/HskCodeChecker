# API 배포용. 루트 requirements.txt 와 **일부러 나눠 둔다** —
# 저쪽에는 streamlit 이 들어 있고, API 서버에는 필요 없는데 설치하면
# 빌드 시간과 디스크만 먹는다. Streamlit 앱은 계속 저쪽을 쓴다.
#
# Render 빌드 명령:  pip install -r api/requirements.txt
fastapi==0.141.1
uvicorn==0.52.3

# fastapi 가 알아서 끌고 오는 것이지만 **여기서 직접 고정한다.**
#
# fastapi 0.141.1 이 요구하는 범위가 `starlette>=0.46.0` 뿐이라 **위쪽이 안
# 막혀 있다.** 그래서 이 파일이 한 글자도 안 바뀌어도 설치하는 시점에 따라
# 다른 버전이 깔린다 — 2026-08-28 에 재 보니 로컬은 1.3.1, 그날 새로 만든
# 환경은 1.6.0 이었다. 위 머리말이 말하는 "로컬에서 되던 앱이 배포에서만
# 깨진다" 가 정확히 이 자리다.
#
# 이 프로젝트는 SSE 로 진행을 흘려보내는데(api/main.py 의 분류_스트림),
# 스트림이 끊겼을 때의 동작이 starlette 판단에 달려 있다. 시험한 버전으로
# 못 박는다.
starlette==1.3.1

# 카탈로그 업로드(multipart/form-data)를 받는 데 필요하다. FastAPI 가
# UploadFile 을 쓸 때만 부르므로 자기 의존성으로 들고 있지 않다.
# **로컬에서는 streamlit 이 끌고 와서 이미 깔려 있다** — 여기 안 적으면
# 로컬에서만 되고 배포에서 500 이 난다.
python-multipart==0.0.32

# 아래 넷은 루트와 **같은 버전이어야 한다.** 두 앱이 같은 Supabase 표에
# 기록하므로, 여기서 버전이 갈리면 같은 입력에 다른 결과가 나올 수 있다.
google-generativeai==0.8.6
numpy==2.4.6
pandas==3.0.5
python-dotenv==1.2.3
psycopg2-binary==2.9.12
